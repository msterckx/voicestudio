# ============================================================
# Runpod persistent storage bootstrap
#
# Normal Pod:
#       /workspace
#
# Serverless:
#       /runpod-volume
#
# We create symlinks so Voice Studio itself continues seeing:
#
#       /opt/Voice-Clone-Studio/samples
#       /opt/Voice-Clone-Studio/models
#       /opt/Voice-Clone-Studio/output
#
# but the actual files live on persistent storage.
# ============================================================

RUN cat > /usr/local/bin/start-voice-studio.sh <<'EOF'
#!/bin/bash

set -euo pipefail


# ------------------------------------------------------------
# Determine persistent volume location
# ------------------------------------------------------------

if [ -d "/runpod-volume" ]; then
    VOLUME_ROOT="/runpod-volume"
elif [ -d "/workspace" ]; then
    VOLUME_ROOT="/workspace"
else
    # Makes the container usable outside Runpod as well.
    VOLUME_ROOT="/data"
    mkdir -p "${VOLUME_ROOT}"
fi


echo "============================================================"
echo "Voice Clone Studio"
echo "Persistent volume: ${VOLUME_ROOT}"
echo "============================================================"


# ------------------------------------------------------------
# Persistent directories
# ------------------------------------------------------------

mkdir -p \
    "${VOLUME_ROOT}/voice-studio/samples" \
    "${VOLUME_ROOT}/voice-studio/models" \
    "${VOLUME_ROOT}/voice-studio/output" \
    "${VOLUME_ROOT}/voice-studio/cache/huggingface" \
    "${VOLUME_ROOT}/voice-studio/cache/modelscope"


# ------------------------------------------------------------
# Hugging Face / ModelScope caches
# ------------------------------------------------------------

export HF_HOME="${VOLUME_ROOT}/voice-studio/cache/huggingface"
export HUGGINGFACE_HUB_CACHE="${HF_HOME}/hub"
export TRANSFORMERS_CACHE="${HF_HOME}/transformers"
export MODELSCOPE_CACHE="${VOLUME_ROOT}/voice-studio/cache/modelscope"


# ------------------------------------------------------------
# Replace Voice Studio data directories with persistent links.
#
# Never replace the application source itself.
# ------------------------------------------------------------

cd /opt/Voice-Clone-Studio


for DIR in samples models output; do

    TARGET="${VOLUME_ROOT}/voice-studio/${DIR}"
    LINK="/opt/Voice-Clone-Studio/${DIR}"

    if [ -L "${LINK}" ]; then
        rm -f "${LINK}"
    elif [ -d "${LINK}" ]; then

        # Preserve anything the repository/container originally
        # supplied before replacing it with the persistent link.
        cp -an "${LINK}/." "${TARGET}/" 2>/dev/null || true

        rm -rf "${LINK}"
    elif [ -e "${LINK}" ]; then
        rm -f "${LINK}"
    fi

    ln -s "${TARGET}" "${LINK}"

done


echo
echo "Samples:"
echo "  ${VOLUME_ROOT}/voice-studio/samples"
echo
echo "Models:"
echo "  ${VOLUME_ROOT}/voice-studio/models"
echo
echo "Outputs:"
echo "  ${VOLUME_ROOT}/voice-studio/output"
echo
echo "Hugging Face cache:"
echo "  ${HF_HOME}"
echo


# Useful diagnostics
python - <<'PY'
import torch

print("Python / Torch startup check")
print("Torch:", torch.__version__)
print("CUDA available:", torch.cuda.is_available())

if torch.cuda.is_available():
    print("GPU:", torch.cuda.get_device_name(0))
    print("CUDA:", torch.version.cuda)
PY


echo
echo "Starting Voice Clone Studio..."
echo

exec python3 voice_clone_studio.py
EOF


RUN chmod +x /usr/local/bin/start-voice-studio.sh


# ============================================================
# Final sanity checks at image-build time
# ============================================================

    python -c "import librosa; print('librosa OK')"
