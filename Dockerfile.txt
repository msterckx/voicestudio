# ============================================================
# Startup script
#
# Storage layouts:
#
# Normal Runpod Pod:
#     /workspace
#
# Serverless with Network Volume:
#     /runpod-volume
#
# Local Docker fallback:
#     /data
#
# Voice Studio itself continues using:
#
# /opt/Voice-Clone-Studio/samples
# /opt/Voice-Clone-Studio/output
# /opt/Voice-Clone-Studio/models
# /opt/Voice-Clone-Studio/datasets
#
# Those are symlinked to persistent storage.
# ============================================================

RUN cat > /usr/local/bin/start-voice-studio.sh <<'EOF'
#!/bin/bash

set -e


echo
echo "============================================================"
echo " Voice Clone Studio"
echo "============================================================"
echo


# ------------------------------------------------------------
# Detect persistent storage location
# ------------------------------------------------------------

if [ -d "/runpod-volume" ]; then

    # Runpod Serverless
    VOLUME_ROOT="/runpod-volume"

elif [ -d "/workspace" ]; then

    # Normal Runpod Pod
    VOLUME_ROOT="/workspace"

else

    # Local Docker fallback
    VOLUME_ROOT="/data"
    mkdir -p "${VOLUME_ROOT}"

fi


VOICE_DATA_ROOT="${VOLUME_ROOT}/voice-studio"


echo "Persistent root : ${VOLUME_ROOT}"
echo "Voice data root : ${VOICE_DATA_ROOT}"
echo


# ------------------------------------------------------------
# Persistent directories
# ------------------------------------------------------------

mkdir -p \
    "${VOICE_DATA_ROOT}/samples" \
    "${VOICE_DATA_ROOT}/output" \
    "${VOICE_DATA_ROOT}/models" \
    "${VOICE_DATA_ROOT}/datasets" \
    "${VOICE_DATA_ROOT}/cache/huggingface" \
    "${VOICE_DATA_ROOT}/cache/modelscope"


# ------------------------------------------------------------
# Persistent Hugging Face / model caches
# ------------------------------------------------------------

export HF_HOME="${VOICE_DATA_ROOT}/cache/huggingface"
export HUGGINGFACE_HUB_CACHE="${HF_HOME}/hub"

export MODELSCOPE_CACHE="${VOICE_DATA_ROOT}/cache/modelscope"


# ------------------------------------------------------------
# Link Voice Studio folders to persistent storage
# ------------------------------------------------------------

link_directory()
{
    NAME="$1"

    SOURCE="${VOICE_DATA_ROOT}/${NAME}"
    TARGET="/opt/Voice-Clone-Studio/${NAME}"

    mkdir -p "${SOURCE}"

    if [ -L "${TARGET}" ]; then

        rm -f "${TARGET}"

    elif [ -d "${TARGET}" ]; then

        # Preserve anything already provided by the repo.
        cp -an "${TARGET}/." "${SOURCE}/" 2>/dev/null || true
        rm -rf "${TARGET}"

    elif [ -e "${TARGET}" ]; then

        rm -f "${TARGET}"

    fi

    ln -s "${SOURCE}" "${TARGET}"
}


link_directory samples
link_directory output
link_directory models
link_directory datasets


# ------------------------------------------------------------
# Show storage configuration
# ------------------------------------------------------------

echo "Persistent links:"
echo

ls -ld \
    /opt/Voice-Clone-Studio/samples \
    /opt/Voice-Clone-Studio/output \
    /opt/Voice-Clone-Studio/models \
    /opt/Voice-Clone-Studio/datasets

echo


# ------------------------------------------------------------
# GPU diagnostics
# ------------------------------------------------------------

python - <<'PY'
import sys
import torch

print("Python:")
print(sys.version)
print()

print("PyTorch:", torch.__version__)
print("PyTorch CUDA runtime:", torch.version.cuda)
print("CUDA available:", torch.cuda.is_available())

if torch.cuda.is_available():

    print("GPU count:", torch.cuda.device_count())
    print("GPU:", torch.cuda.get_device_name(0))

else:

    print()
    print("WARNING: CUDA is not available.")
PY


echo
echo "Reference voice samples:"
echo

find "${VOICE_DATA_ROOT}/samples" \
    -maxdepth 1 \
    -type f \
    -printf '  %f\n' \
    2>/dev/null || true


echo
echo "============================================================"
echo " Starting Voice Clone Studio"
echo " Port: 7860"
echo "============================================================"
echo


cd /opt/Voice-Clone-Studio

exec python voice_clone_studio.py

EOF


RUN chmod +x /usr/local/bin/start-voice-studio.sh
